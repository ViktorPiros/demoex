(
    id serial,
    organization_type text,
    name text UNIQUE,
    ceo text,
    email text,
    phone text,
    address text,
    tax_id text,
    rating integer,
    PRIMARY KEY (id)
);

(
    id serial,
    product_type text,
    name text NOT NULL,
    article text NOT NULL,
